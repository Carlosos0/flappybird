469// --- LÓGICA DO JOGO ---
470void gamecontroller::Logic(){
471 if (!gameBird) return;
472
473 if (gameInput.getJumpInput() && gameBird->getAlive()) {
474 gameBird->JumpAct();
475 }
476 gameInput.ResetJumpState();
477
478 if (gameBird->getAlive()) {
479 gameBird->Movement();
480 gamePControl->PipeGenerator();
481 gamePControl->UpdatePos();
482 gamePControl->DeletePipes();
483 gameCrash->checkCrash(*gameBird, *gamePControl);
484 gameScore->updateScore(*gameBird, *gamePControl);
485 } else {
486 if (!gameCrash->getHitFloor()) {
487 gameBird->Movement();
488 gameCrash->checkCrash(*gameBird, *gamePControl);
489 }
490 if (gameCrash->getHitFloor()) {
491
492 gameBird->setMS(vetor(0, 0));
493
494 if (deathTime == 0.0) {
495 deathTime = al_get_time();
496
497 int finalScore = gameScore->getCurrentScore();
498 if (!currentPlayerNickname.empty()) {
499 int bestScore = playerManager->getHighScore(currentPlayerNickname);
500 if (finalScore > bestScore) {
501 newRecord = true;
502 }
503 playerManager->updatePlayerScore(currentPlayerNickname, finalScore);
504 }
505 }
506 else if (al_get_time() - deathTime > 0.3) {
507 currentState = GameState::GameOver;
508 }
509 }
510 }
511}
